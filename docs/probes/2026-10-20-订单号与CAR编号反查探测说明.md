## O1 各编号列的写法形态

每列只列出现最多的 15 种形态；「（空）」是 NULL 或空串。

```sql
WITH v AS (
    SELECT '国内生产订单.订单号' AS "列", o."OrderNumber"::text AS x FROM "ZCTracking_ProblemOrders" o
    UNION ALL SELECT '国内小板订单.订单号', x."OrderNumber"::text FROM "ZCTracking_XBOrders" x
    UNION ALL SELECT '国内其他在线订单.订单号', t."OrderNo"::text FROM "ZCTracking_OtherOrder" t
    UNION ALL SELECT '国内主表.CAR编号', m."CAR"::text FROM "ZCTracking_ProblemList" m
    UNION ALL SELECT '国内主表.在线退货单号', m."OReturnOrderNo"::text FROM "ZCTracking_ProblemList" m
    UNION ALL SELECT '国内主表.在库退货单号', m."SReturnOrderNo"::text FROM "ZCTracking_ProblemList" m
    UNION ALL SELECT '存货档案.料号', i."cInvCode"::text FROM "Inventory" i
    UNION ALL SELECT '海外发货单.PO', d."PO"::text FROM "OverseasProblem_DeliveryOrder" d
    UNION ALL SELECT '海外发货单.发货单号', d."OrderNumber"::text FROM "OverseasProblem_DeliveryOrder" d
), f AS (
    SELECT "列", NULLIF(btrim(x), '') AS x FROM v
), s AS (
    SELECT "列", x,
        CASE WHEN x IS NULL THEN '（空）'
             ELSE regexp_replace(regexp_replace(regexp_replace(x, '[0-9]', '9', 'g'),
                                                '[A-Za-z]', 'A', 'g'),
                                 '[^ -~]', '中', 'g') END AS "形态"
    FROM f
), g AS (
    SELECT "列", "形态", count(*) AS n, count(DISTINCT x) AS nd,
           sum(count(*)) OVER (PARTITION BY "列") AS total,
           row_number() OVER (PARTITION BY "列" ORDER BY count(*) DESC) AS rk
    FROM s
    GROUP BY "列", "形态"
)
SELECT "列", total AS "该列总行数", "形态", n AS "行数", nd AS "去重数"
FROM g
WHERE rk <= 15
ORDER BY "列", n DESC;
```

## O2 订单号前两位与年份

只看 6 位纯数字的订单号。小板订单没有发料日期，只比建单年份。

```sql
WITH o AS (
    SELECT '生产订单' AS "表", btrim(o."OrderNumber"::text) AS n, o."IssueDate" AS d, o."ProblemId" AS p
    FROM "ZCTracking_ProblemOrders" o
    UNION ALL
    SELECT '小板订单', btrim(x."OrderNumber"::text), NULL::timestamp, x."ProblemId"
    FROM "ZCTracking_XBOrders" x
)
SELECT o."表", left(o.n, 2) AS "前两位", count(*) AS "行数",
       min(extract(year FROM o.d))::int AS "最早发料年",
       max(extract(year FROM o.d))::int AS "最晚发料年",
       min(extract(year FROM m."AddDate"))::int AS "最早建单年",
       max(extract(year FROM m."AddDate"))::int AS "最晚建单年"
FROM o
JOIN "ZCTracking_ProblemList" m ON m."ID" = o.p
WHERE o.n ~ '^[0-9]{6}$'
GROUP BY o."表", left(o.n, 2)
ORDER BY o."表", left(o.n, 2);
```

## O3 一个订单号对应几张单、会不会与别的编号撞

三种国内订单一起看。「未删除」按国内主表是否删除判断。

```sql
WITH o AS (
    SELECT '生产订单' AS t, btrim("OrderNumber"::text) AS n, "ProblemId" AS p FROM "ZCTracking_ProblemOrders"
    UNION ALL SELECT '小板订单', btrim("OrderNumber"::text), "ProblemId" FROM "ZCTracking_XBOrders"
    UNION ALL SELECT '其他在线订单', btrim("OrderNo"::text), "ProblemId" FROM "ZCTracking_OtherOrder"
), g AS (
    SELECT o.n,
           count(DISTINCT o.t) AS nt,
           count(DISTINCT o.p) FILTER (WHERE COALESCE(m."IsDelete"::int, 0) = 0) AS np_live,
           bool_or(o.t = '生产订单') AS in_prod,
           bool_or(o.t = '小板订单') AS in_xb,
           bool_or(o.t = '其他在线订单') AS in_other
    FROM o
    JOIN "ZCTracking_ProblemList" m ON m."ID" = o.p
    WHERE NULLIF(o.n, '') IS NOT NULL
    GROUP BY o.n
), inv AS (
    SELECT DISTINCT btrim("cInvCode"::text) AS n FROM "Inventory"
), car AS (
    SELECT DISTINCT btrim("CAR"::text) AS n FROM "ZCTracking_ProblemList"
    WHERE NULLIF(btrim("CAR"::text), '') IS NOT NULL
), g2 AS (
    SELECT g.*, inv.n IS NOT NULL AS is_inv, car.n IS NOT NULL AS is_car
    FROM g
    LEFT JOIN inv ON inv.n = g.n
    LEFT JOIN car ON car.n = g.n
)
SELECT '订单号（去重）' AS "项", count(*) AS "个数" FROM g2
UNION ALL SELECT '同时出现在两种及以上订单里', count(*) FILTER (WHERE nt > 1) FROM g2
UNION ALL SELECT '  其中生产订单与小板订单都有', count(*) FILTER (WHERE in_prod AND in_xb) FROM g2
UNION ALL SELECT '  其中生产订单与其他在线订单都有', count(*) FILTER (WHERE in_prod AND in_other) FROM g2
UNION ALL SELECT '只挂在已删除的单上', count(*) FILTER (WHERE np_live = 0) FROM g2
UNION ALL SELECT '对应 1 张未删除的单', count(*) FILTER (WHERE np_live = 1) FROM g2
UNION ALL SELECT '对应 2–5 张未删除的单', count(*) FILTER (WHERE np_live BETWEEN 2 AND 5) FROM g2
UNION ALL SELECT '对应 6 张以上未删除的单', count(*) FILTER (WHERE np_live > 5) FROM g2
UNION ALL SELECT '单个订单号最多对应几张未删除的单', COALESCE(max(np_live), 0) FROM g2
UNION ALL SELECT '与某个料号写法相同', count(*) FILTER (WHERE is_inv) FROM g2
UNION ALL SELECT '与某个 CAR 编号写法相同', count(*) FILTER (WHERE is_car) FROM g2;
```

## O4 CAR 编号的分布

物料问题的问题类型为 3。

```sql
WITH c AS (
    SELECT NULLIF(btrim("CAR"::text), '') AS car, "ProblemType"::int AS pt,
           COALESCE("IsDelete"::int, 0) AS del
    FROM "ZCTracking_ProblemList"
), k AS (
    SELECT car, count(*) FILTER (WHERE del = 0) AS n_live,
           count(*) FILTER (WHERE del = 0 AND pt = 3) AS n_live_mat
    FROM c WHERE car IS NOT NULL GROUP BY car
)
SELECT '有 CAR 编号的单' AS "项", count(*) AS "个数" FROM c WHERE car IS NOT NULL
UNION ALL SELECT '  其中未删除的物料问题', count(*) FROM c WHERE car IS NOT NULL AND del = 0 AND pt = 3
UNION ALL SELECT '  其中未删除的其他类型', count(*) FROM c WHERE car IS NOT NULL AND del = 0 AND pt <> 3
UNION ALL SELECT '  其中已删除', count(*) FROM c WHERE car IS NOT NULL AND del = 1
UNION ALL SELECT 'CAR 编号（去重）', count(*) FROM k
UNION ALL SELECT '被 2 张及以上未删除单共用的 CAR 编号', count(*) FROM k WHERE n_live >= 2
UNION ALL SELECT '被 2 张及以上未删除物料问题共用的 CAR 编号', count(*) FROM k WHERE n_live_mat >= 2
UNION ALL SELECT '单个 CAR 编号最多被几张未删除单共用', COALESCE(max(n_live), 0) FROM k;
```

## O5 料号与 PN 编码会不会撞（补充，用于 G28）

料号（存货档案）与 PN 编码（PN 主数据）都是 15 位数字。看两套号是否重叠，以及问题单上实际用到的号里有多少重叠。先确认镜像库里有 `NewProduct` 表（O0 没查它）：如果下面报「表不存在」，把报错发回即可。

「问题单上用到的料号」取国内、巴西主表的料号；「问题单上用到的 PN」取国内生产订单上的 PN 编码与海外主表的 PN（逗号多值逐段拆开）。海外物料清单是 JSON，不在内。

```sql
WITH inv AS (
    SELECT DISTINCT btrim("cInvCode"::text) AS n FROM "Inventory"
    WHERE btrim("cInvCode"::text) ~ '^[0-9]{15}$'
), pn AS (
    SELECT DISTINCT btrim("PartNumber"::text) AS n FROM "NewProduct"
    WHERE btrim("PartNumber"::text) ~ '^[0-9]{15}$'
), used_inv AS (
    SELECT btrim("CInvCode"::text) AS n FROM "ZCTracking_ProblemList"
    UNION SELECT btrim("cInvCode"::text) FROM "BrazilProblemList"
), used_pn AS (
    SELECT btrim("PartNumber"::text) AS n FROM "ZCTracking_ProblemOrders"
    UNION SELECT btrim(x) FROM "OverseasProblemDetail" o,
                 regexp_split_to_table(COALESCE(o."PNList"::text, ''), ',') AS x
), used AS (
    SELECT n FROM used_inv WHERE n ~ '^[0-9]{15}$'
    UNION SELECT n FROM used_pn WHERE n ~ '^[0-9]{15}$'
)
SELECT '料号（15 位数字，去重）' AS "项", count(*) AS "个数" FROM inv
UNION ALL SELECT 'PN 编码（15 位数字，去重）', count(*) FROM pn
UNION ALL SELECT '既是料号又是 PN 编码', count(*) FROM inv JOIN pn USING (n)
UNION ALL SELECT '问题单上用到的 15 位数字（去重）', count(*) FROM used
UNION ALL SELECT '  其中既是料号又是 PN 编码', count(*) FROM used JOIN inv USING (n) JOIN pn USING (n);
```

**O5b 前三位的分布**（看两套号能不能靠开头几位区分；只输出前三位与计数）

```sql
SELECT '料号' AS "类", left(btrim("cInvCode"::text), 3) AS "前三位", count(*) AS "个数"
FROM "Inventory" WHERE btrim("cInvCode"::text) ~ '^[0-9]{15}$'
GROUP BY 2
UNION ALL
SELECT 'PN 编码', left(btrim("PartNumber"::text), 3), count(*)
FROM "NewProduct" WHERE btrim("PartNumber"::text) ~ '^[0-9]{15}$'
GROUP BY 2
ORDER BY 1, 3 DESC;
```
