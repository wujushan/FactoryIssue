## X1 国内：分析类环节当前退回、不显示的单（镜像库，可发回）

每张单每个分析类环节（编码 3、4、6、10、11、13、14；35 预防对策确认总是显示，不数）取最新一条，再按单张单详情的显示条件判断这个环节现在显示不显示（国内「单张单详情各环节怎么显示」）：本类型有这一环节、也有这一条记录，只因当前状态退回到它之前而不显示，记为「退回、当前不显示」。「其中有文字」是意见、内容（工厂制程分析另加结论）剥掉标签后不为空的，这些会被文字查找找到。

```sql
WITH p AS (
    SELECT "ID" AS id, "Status"::int AS s, "ProblemType"::int AS t,
           coalesce("OldProblemType"::int, 0) AS old, coalesce("ZLConfirmResult"::int, 0) AS zl
    FROM "ZCTracking_ProblemList"
    WHERE coalesce("IsDelete"::int, 0) = 0
),
l AS (
    SELECT DISTINCT ON (l."ProblemId", l."Step"::int)
           l."ProblemId" AS id, l."Step"::int AS code,
           btrim(regexp_replace(coalesce(l."Opinon", '') || coalesce(l."Measures", '')
                 || CASE WHEN l."Step"::int = 6 THEN coalesce(l."Result"::text, '') ELSE '' END,
                 '<[^>]+>', '', 'g')) <> '' AS has_text
    FROM "ZCTracking_ProblemSteplog" l
    WHERE l."Step"::int IN (3, 4, 6, 10, 11, 13, 14)
    ORDER BY l."ProblemId", l."Step"::int, l."ID" DESC
),
r AS (
    SELECT p.id, l.code, l.has_text,
           CASE l.code
               WHEN 3 THEN p.s <= 2
               WHEN 14 THEN (p.t = 3 OR p.old = 3) AND p.s = 11
               WHEN 4 THEN (p.t = 4 AND p.s <= 3) OR (p.t <> 4 AND p.old = 4 AND p.zl = 0 AND p.s = 3)
               WHEN 10 THEN p.t = 3 AND p.s <= 8
               WHEN 11 THEN p.t = 3 AND p.s <= 9
               WHEN 13 THEN p.t = 3 AND p.s <= 10
               WHEN 6 THEN p.t = 5 AND p.s <= 5
           END AS returned
    FROM p
    JOIN l ON l.id = p.id
)
SELECT code::text AS "环节编码",
       count(*) AS "有这一环节记录的单",
       count(*) FILTER (WHERE returned) AS "退回、当前不显示",
       count(*) FILTER (WHERE returned AND has_text) AS "其中有文字"
FROM r
GROUP BY code
UNION ALL
SELECT '合计（按单去重）', count(DISTINCT id), count(DISTINCT id) FILTER (WHERE returned),
       count(DISTINCT id) FILTER (WHERE returned AND has_text)
FROM r
ORDER BY 1;
```

## X2 海外：当前环节退回后还没重新提交的单（镜像库，可发回）

海外按当前状态对应的环节判断（海外「单张单详情」被驳回过的环节）：单上有驳回记录，当前状态这个环节有过记录，而它最近一条记录在最近一次驳回之前，就是退回后还没重新提交。

```sql
WITH p AS (
    SELECT "Id" AS id,
           CASE "Status"::int
               WHEN 1 THEN '指定责任人' WHEN 2 THEN '责任人确认' WHEN 3 THEN '其他问题确认'
               WHEN 4 THEN 'IQC意见' WHEN 5 THEN '工厂制程分析' WHEN 6 THEN 'IQC确认'
               WHEN 7 THEN '填写退货单号' WHEN 8 THEN 'SQE跟进' WHEN 9 THEN '设计更改'
               WHEN 10 THEN '设计变更后SQE确认' WHEN 30 THEN '效果确认' WHEN 31 THEN '结案确认'
               WHEN 32 THEN '已完成'
           END AS label
    FROM "OverseasProblemDetail"
    WHERE "IsDelete"::int = 0
),
rj AS (
    SELECT "ProblemId" AS id, max("Id") AS last_reject
    FROM "OverseasProblem_StepLog"
    WHERE "StepStr" = '驳回'
    GROUP BY 1
),
st AS (
    SELECT p.id, max(l."Id") AS last_stage
    FROM p
    JOIN "OverseasProblem_StepLog" l
      ON l."ProblemId" = p.id
     AND (l."StepStr" = p.label OR (p.label = 'IQC意见' AND l."StepStr" IN ('IQC意见', 'IQC填写意见')))
    GROUP BY p.id
)
SELECT count(*) AS "未删除的单",
       count(rj.id) AS "有驳回记录的单",
       count(*) FILTER (WHERE st.last_stage IS NOT NULL AND rj.last_reject > st.last_stage) AS "退回后还没重新提交",
       count(*) FILTER (WHERE st.last_stage IS NOT NULL AND rj.last_reject > st.last_stage
                        AND p.label IN ('责任人确认', 'IQC意见', '工厂制程分析', 'SQE跟进', '设计变更后SQE确认'))
           AS "其中是分析类环节"
FROM p
LEFT JOIN rj ON rj.id = p.id
LEFT JOIN st ON st.id = p.id;
```
